# Building a Key-Value Store Device

A beginner's guide to creating your first HyperBEAM device

---

## What You'll Build

A personal key-value store device that lets users store, retrieve, and delete data. By the end, you'll have a working device with these endpoints:

```
POST /~kv@1.0/set?key=mykey        Store value with key
GET  /~kv@1.0/get?key=mykey        Retrieve value
GET  /~kv@1.0/keys                 List all keys
DELETE /~kv@1.0/delete?key=mykey   Delete key
GET  /~kv@1.0/info                 Device metadata
```

---

1. **Device Structure** - The anatomy of a HyperBEAM device module

## Part 1: Device Anatomy

Every HyperBEAM device is an Erlang module with functions that handle requests. The simplest device has one function: `info/3`.

### The Minimum Device

Create `HyperBEAM/src/dev_kv.erl`:

```erlang
-module(dev_kv).
-export([info/3]).
-include("include/hb.hrl").

%% Device metadata - called for /~kv@1.0/info
info(_M1, _M2, _Opts) ->
    {ok, #{
        <<"name">> => <<"kv">>,
        <<"version">> => <<"1.0">>,
        <<"description">> => <<"Personal Key-Value Store">>
    }}.
```

**Understanding the function signature:**

```erlang
info(M1, M2, Opts) ->
    %% M1 = Base message (device configuration, process state)
    %% M2 = Request message (user's HTTP request converted to message)
    %% Opts = Execution options (store, cache settings, etc.)
    {ok, Response}.
```

Every device function returns `{ok, Result}` or `{error, Reason}`.

### The Response Flow

When a user requests `/~kv@1.0/info`:

```
HTTP GET /~kv@1.0/info
    ↓
hb_http_server converts to message
    ↓
hb_ao:resolve finds device kv@1.0
    ↓
dev_kv:info/3 called
    ↓
Response returned as HTTP
```

## Part 2: Reading Request Data with hb_maps:get

:::warning[Important API Note]
Use `hb_maps:get/4` to read parameters from the request message (M2). Do NOT use `hb_ao:get` for this purpose as it performs AO-Core resolution which can transform values unexpectedly.
:::

The `hb_maps:get/4` function reads values directly from messages:

### Basic Usage

```erlang
%% Read a single key with default
Value = hb_maps:get(<<"key">>, M2, not_found, Opts),

%% Check if key exists
case hb_maps:get(<<"key">>, M2, not_found, Opts) of
    not_found -> handle_missing();
    Value -> handle_value(Value)
end.
```

### Implementing GET

Add the `get` function to retrieve stored values:

```erlang
-export([info/3, get/3]).

%% GET /~kv@1.0/get?key=mykey
get(M1, M2, Opts) ->
    %% Read the key from request using hb_maps:get (NOT hb_ao:get)
    Key = hb_maps:get(<<"key">>, M2, not_found, Opts),
    case Key of
        not_found ->
            {error, #{<<"error">> => <<"key_required">>}};
        _ ->
            %% Read value from private state
            %% hb_private:get(Key, Msg, Default, Opts)
            case hb_private:get(Key, M1, not_found, Opts) of
                not_found ->
                    {error, #{<<"error">> => <<"not_found">>, <<"key">> => Key}};
                Value ->
                    {ok, #{<<"key">> => Key, <<"value">> => Value}}
            end
    end.
```

## Part 3: Private State with hb_private

`hb_private` stores state that's invisible to external callers. It's perfect for device-internal data.

### How Private State Works

```
Message (visible to user):
#{
    <<"device">> => <<"kv@1.0">>,
    <<"public">> => <<"data">>
}

Private State (hidden):
#{
    <<"my-secret">> => <<"password">>,
    <<"internal">> => <<"state">>
}
```

### Private State API

```erlang
%% Get a specific key from private state
Value = hb_private:get(Key, M1, Default, Opts),
%% Returns Value or Default if not found

%% Get all private state as a map
Priv = hb_private:from_message(M1),
%% Priv = #{<<"key1">> => Value1, ...}

%% Get specific private key
Value = hb_private:get(M1, <<"key">>, Opts),
%% Value = ... or not_found

%% Set private state (returns updated message)
M1Updated = hb_private:set(M1, #{<<"key">> => <<"value">>}, Opts).
```

### Implementing SET

Add the `set` function to store values:

```erlang
-export([info/3, get/3, set/3]).

%% POST /~kv@1.0/set?key=mykey (with body as value)
set(M1, M2, Opts) ->
    Key = hb_maps:get(<<"key">>, M2, not_found, Opts),
    Value = hb_maps:get(<<"value">>, M2, not_found, Opts),
    case {Key, Value} of
        {not_found, _} ->
            {error, #{<<"error">> => <<"key_required">>}};
        {_, not_found} ->
            {error, #{<<"error">> => <<"value_required">>}};
        {K, V} ->
            %% hb_private:set returns updated message
            M1Updated = hb_private:set(M1, K, V, Opts),
            {ok, M1Updated#{<<"status">> => <<"stored">>, <<"key">> => K}}
    end.
```

## Part 4: Persisting with hb_cache

Private state lives in memory. For persistence, use `hb_cache` which stores data by content hash.

### Cache Basics

```erlang
%% Write to cache (returns content-addressed ID)
{ok, ID} = hb_cache:write(Data, Opts),

%% Read from cache
{ok, Data} = hb_cache:read(ID, Opts),
not_found = hb_cache:read(<<"nonexistent">>, Opts).
```

### Adding Persistence

Let's modify our device to persist state to cache:

```erlang
-module(dev_kv).
-export([info/3, get/3, set/3, delete/3, keys/3]).
-include("include/hb.hrl").

%% Key for storing our state ID in private
-define(STATE_KEY, <<"kv-state-id">>).

%% === PUBLIC API ===

info(_M1, _M2, _Opts) ->
    {ok, #{
        <<"name">> => <<"kv">>,
        <<"version">> => <<"1.0">>,
        <<"description">> => <<"Personal Key-Value Store with Persistence">>
    }}.

get(M1, M2, Opts) ->
    Key = hb_maps:get(<<"key">>, M2, not_found, Opts),
    case Key of
        not_found ->
            {error, #{<<"error">> => <<"key_required">>}};
        _ ->
            case hb_private:get(Key, M1, not_found, Opts) of
                not_found ->
                    {error, #{<<"error">> => <<"not_found">>, <<"key">> => Key}};
                Value ->
                    {ok, #{<<"key">> => Key, <<"value">> => Value}}
            end
    end.

set(M1, M2, Opts) ->
    case hb_ao:get(<<"key">>, M2, Opts) of
        not_found ->
            {error, #{<<"status">> => 400, <<"error">> => <<"Missing 'key'">>}};
        Key ->
            Value = hb_ao:get(<<"value">>, M2, Opts, <<>>),
            State = load_state(M1, Opts),
            NewState = maps:put(Key, Value, State),
            M1Updated = save_state(M1, NewState, Opts),
            {ok, hb_ao:set(M1Updated, #{
                <<"status">> => <<"stored">>,
                <<"key">> => Key
            }, Opts)}
    end.

delete(M1, M2, Opts) ->
    case hb_ao:get(<<"key">>, M2, Opts) of
        not_found ->
            {error, #{<<"status">> => 400, <<"error">> => <<"Missing 'key'">>}};
        Key ->
            State = load_state(M1, Opts),
            case maps:is_key(Key, State) of
                false ->
                    {error, #{<<"status">> => 404, <<"error">> => <<"Key not found">>}};
                true ->
                    NewState = maps:remove(Key, State),
                    M1Updated = save_state(M1, NewState, Opts),
                    {ok, hb_ao:set(M1Updated, #{
                        <<"status">> => <<"deleted">>,
                        <<"key">> => Key
                    }, Opts)}
            end
    end.

keys(M1, _M2, Opts) ->
    State = load_state(M1, Opts),
    Keys = maps:keys(State),
    {ok, #{<<"keys">> => Keys, <<"count">> => length(Keys)}}.

%% === INTERNAL HELPERS ===

%% Load state from cache
load_state(M1, Opts) ->
    case hb_private:get(M1, ?STATE_KEY, Opts) of
        not_found ->
            #{};
        StateID ->
            case hb_cache:read(StateID, Opts) of
                {ok, State} ->
                    hb_cache:ensure_all_loaded(State, Opts);
                not_found ->
                    #{}
            end
    end.

%% Save state to cache and update private reference
save_state(M1, State, Opts) ->
    {ok, StateID} = hb_cache:write(State, Opts),
    hb_private:set(M1, #{?STATE_KEY => StateID}, Opts).
```

## Part 5: Storage Backends with hb_store

HyperBEAM supports multiple storage backends. The cache uses whatever store is configured in options.

### Available Backends

| Backend | Module | Best For |
|---------|--------|----------|
| Filesystem | `hb_store_fs` | Development, simple setups |
| LMDB | `hb_store_lmdb` | Production, fast reads |
| RocksDB | `hb_store_rocksdb` | Write-heavy workloads |
| LRU Cache | `hb_store_lru` | Wrapping any store with memory cache |

### Store Chains

You can chain stores for tiered storage:

```erlang
%% LRU cache in front of LMDB
Opts = #{
    store => hb_store_lru:new(#{
        max_size => 1000,
        store => hb_store_lmdb:open("data/lmdb")
    })
}.
```

### Checking Store in Device

```erlang
%% Get the configured store
Store = hb_opts:get(store, no_viable_store, Opts),

%% Use store directly (rarely needed)
{ok, Data} = hb_store:read(Store, Key),
ok = hb_store:write(Store, Key, Data).
```

## Part 7: Testing

### Erlang Unit Tests

Add tests directly in your module:

```erlang
-ifdef(TEST).
-include_lib("eunit/include/eunit.hrl").

basic_operations_test() ->
    %% Setup
    Store = hb_store_fs:new("test_data"),
    hb_store:reset(Store),
    Opts = #{store => Store},
    M1 = #{<<"device">> => <<"kv@1.0">>},

    %% Test info
    {ok, Info} = info(M1, #{}, Opts),
    ?assertEqual(<<"kv">>, maps:get(<<"name">>, Info)),

    %% Test set
    M2Set = #{<<"key">> => <<"test-key">>, <<"value">> => <<"test-value">>},
    {ok, SetResult} = set(M1, M2Set, Opts),
    ?assertEqual(<<"stored">>, hb_ao:get(<<"status">>, SetResult, Opts)),

    %% Get the updated M1 with state
    M1Updated = SetResult,

    %% Test get
    M2Get = #{<<"key">> => <<"test-key">>},
    {ok, GetResult} = get(M1Updated, M2Get, Opts),
    ?assertEqual(<<"test-value">>, maps:get(<<"value">>, GetResult)),

    %% Test keys
    {ok, KeysResult} = keys(M1Updated, #{}, Opts),
    ?assertEqual(1, maps:get(<<"count">>, KeysResult)),

    %% Test delete
    {ok, DeleteResult} = delete(M1Updated, M2Get, Opts),
    ?assertEqual(<<"deleted">>, hb_ao:get(<<"status">>, DeleteResult, Opts)),

    %% Cleanup
    hb_store:reset(Store),
    ok.

missing_key_test() ->
    Opts = #{store => hb_store_fs:new("test_data")},
    M1 = #{<<"device">> => <<"kv@1.0">>},
    M2 = #{},

    {error, Error} = get(M1, M2, Opts),
    ?assertEqual(400, maps:get(<<"status">>, Error)).

-endif.
```

### WAO SDK Integration Tests

Create `test/test-kv.js`:

```javascript
import assert from "assert"
import { describe, it, before, after } from "node:test"
import { HyperBEAM } from "wao"

describe("Key-Value Store Device", function () {
    let hbeam, hb

    before(async () => {
        hbeam = await new HyperBEAM({
            cwd: "../HyperBEAM",
            reset: true
        }).ready()
        hb = hbeam.hb
    })

    after(async () => hbeam.kill())

    it("should return device info", async () => {
        const info = await hb.get("/~kv@1.0/info")
        assert.equal(info.name, "kv")
        assert.equal(info.version, "1.0")
    })

    it("should store and retrieve values", async () => {
        // Store a value
        const setRes = await hb.post("/~kv@1.0/set", {
            key: "greeting",
            value: "Hello, HyperBEAM!"
        })
        assert.equal(setRes.status, "stored")

        // Retrieve the value
        const getRes = await hb.get("/~kv@1.0/get?key=greeting")
        assert.equal(getRes.value, "Hello, HyperBEAM!")
    })

    it("should list all keys", async () => {
        // Store multiple values
        await hb.post("/~kv@1.0/set", { key: "key1", value: "value1" })
        await hb.post("/~kv@1.0/set", { key: "key2", value: "value2" })

        // List keys
        const keysRes = await hb.get("/~kv@1.0/keys")
        assert.ok(keysRes.keys.includes("key1"))
        assert.ok(keysRes.keys.includes("key2"))
        assert.ok(keysRes.count >= 2)
    })

    it("should delete keys", async () => {
        // Store a value
        await hb.post("/~kv@1.0/set", { key: "temp", value: "temporary" })

        // Delete it
        const deleteRes = await hb.post("/~kv@1.0/delete?key=temp")
        assert.equal(deleteRes.status, "deleted")

        // Verify it's gone
        try {
            await hb.get("/~kv@1.0/get?key=temp")
            assert.fail("Should have thrown")
        } catch (e) {
            assert.equal(e.status, 404)
        }
    })

    it("should return 400 for missing key parameter", async () => {
        try {
            await hb.get("/~kv@1.0/get")
            assert.fail("Should have thrown")
        } catch (e) {
            assert.equal(e.status, 400)
        }
    })

    it("should return 404 for nonexistent key", async () => {
        try {
            await hb.get("/~kv@1.0/get?key=nonexistent")
            assert.fail("Should have thrown")
        } catch (e) {
            assert.equal(e.status, 404)
        }
    })
})
```

## Complete Code

Here's the final `dev_kv.erl`:

```erlang
%%%-------------------------------------------------------------------
%%% @doc Key-Value Store Device
%%%
%%% A personal key-value store with persistent storage.
%%%
%%% API:
%%%   GET  /~kv@1.0/info              Device metadata
%%%   GET  /~kv@1.0/get?key=KEY       Get value
%%%   POST /~kv@1.0/set?key=KEY       Set value (body = value)
%%%   POST /~kv@1.0/delete?key=KEY    Delete key
%%%   GET  /~kv@1.0/keys              List all keys
%%%
%%% @end
%%%-------------------------------------------------------------------
-module(dev_kv).
-export([info/3, get/3, set/3, delete/3, keys/3]).
-include("include/hb.hrl").

-define(STATE_KEY, <<"kv-state-id">>).

%%====================================================================
%% Public API
%%====================================================================

%% @doc Device metadata
info(_M1, _M2, _Opts) ->
    {ok, #{
        <<"name">> => <<"kv">>,
        <<"version">> => <<"1.0">>,
        <<"description">> => <<"Personal Key-Value Store with Persistence">>,
        <<"author">> => <<"HyperBEAM Book">>
    }}.

%% @doc Get value by key
get(M1, M2, Opts) ->
    case hb_ao:get(<<"key">>, M2, Opts) of
        not_found ->
            {error, #{<<"status">> => 400, <<"error">> => <<"Missing 'key' parameter">>}};
        Key ->
            State = load_state(M1, Opts),
            case maps:get(Key, State, not_found) of
                not_found ->
                    {error, #{<<"status">> => 404, <<"error">> => <<"Key not found">>}};
                Value ->
                    {ok, #{<<"key">> => Key, <<"value">> => Value}}
            end
    end.

%% @doc Set key to value
set(M1, M2, Opts) ->
    case hb_ao:get(<<"key">>, M2, Opts) of
        not_found ->
            {error, #{<<"status">> => 400, <<"error">> => <<"Missing 'key' parameter">>}};
        Key ->
            Value = hb_ao:get(<<"value">>, M2, Opts, <<>>),
            State = load_state(M1, Opts),
            NewState = maps:put(Key, Value, State),
            M1Updated = save_state(M1, NewState, Opts),
            {ok, hb_ao:set(M1Updated, #{
                <<"status">> => <<"stored">>,
                <<"key">> => Key
            }, Opts)}
    end.

%% @doc Delete key
delete(M1, M2, Opts) ->
    case hb_ao:get(<<"key">>, M2, Opts) of
        not_found ->
            {error, #{<<"status">> => 400, <<"error">> => <<"Missing 'key' parameter">>}};
        Key ->
            State = load_state(M1, Opts),
            case maps:is_key(Key, State) of
                false ->
                    {error, #{<<"status">> => 404, <<"error">> => <<"Key not found">>}};
                true ->
                    NewState = maps:remove(Key, State),
                    M1Updated = save_state(M1, NewState, Opts),
                    {ok, hb_ao:set(M1Updated, #{
                        <<"status">> => <<"deleted">>,
                        <<"key">> => Key
                    }, Opts)}
            end
    end.

%% @doc List all keys
keys(M1, _M2, Opts) ->
    State = load_state(M1, Opts),
    Keys = maps:keys(State),
    {ok, #{<<"keys">> => Keys, <<"count">> => length(Keys)}}.

%%====================================================================
%% Internal Functions
%%====================================================================

%% @private Load state from cache
load_state(M1, Opts) ->
    case hb_private:get(M1, ?STATE_KEY, Opts) of
        not_found ->
            #{};
        StateID ->
            case hb_cache:read(StateID, Opts) of
                {ok, State} ->
                    hb_cache:ensure_all_loaded(State, Opts);
                not_found ->
                    #{}
            end
    end.

%% @private Save state to cache
save_state(M1, State, Opts) ->
    {ok, StateID} = hb_cache:write(State, Opts),
    hb_private:set(M1, #{?STATE_KEY => StateID}, Opts).

%%====================================================================
%% Tests
%%====================================================================

-ifdef(TEST).
-include_lib("eunit/include/eunit.hrl").

setup_test_env() ->
    Store = hb_store_fs:new("test_kv_data"),
    hb_store:reset(Store),
    #{store => Store}.

info_test() ->
    {ok, Info} = info(#{}, #{}, #{}),
    ?assertEqual(<<"kv">>, maps:get(<<"name">>, Info)),
    ?assertEqual(<<"1.0">>, maps:get(<<"version">>, Info)).

set_get_test() ->
    Opts = setup_test_env(),
    M1 = #{<<"device">> => <<"kv@1.0">>},

    %% Set value
    {ok, SetRes} = set(M1, #{<<"key">> => <<"foo">>, <<"value">> => <<"bar">>}, Opts),
    ?assertEqual(<<"stored">>, hb_ao:get(<<"status">>, SetRes, Opts)),

    %% Get value
    {ok, GetRes} = get(SetRes, #{<<"key">> => <<"foo">>}, Opts),
    ?assertEqual(<<"bar">>, maps:get(<<"value">>, GetRes)).

delete_test() ->
    Opts = setup_test_env(),
    M1 = #{<<"device">> => <<"kv@1.0">>},

    %% Set then delete
    {ok, SetRes} = set(M1, #{<<"key">> => <<"temp">>, <<"value">> => <<"data">>}, Opts),
    {ok, DelRes} = delete(SetRes, #{<<"key">> => <<"temp">>}, Opts),
    ?assertEqual(<<"deleted">>, hb_ao:get(<<"status">>, DelRes, Opts)),

    %% Verify gone
    {error, _} = get(DelRes, #{<<"key">> => <<"temp">>}, Opts).

keys_test() ->
    Opts = setup_test_env(),
    M1 = #{<<"device">> => <<"kv@1.0">>},

    {ok, M2} = set(M1, #{<<"key">> => <<"a">>, <<"value">> => <<"1">>}, Opts),
    {ok, M3} = set(M2, #{<<"key">> => <<"b">>, <<"value">> => <<"2">>}, Opts),

    {ok, KeysRes} = keys(M3, #{}, Opts),
    ?assertEqual(2, maps:get(<<"count">>, KeysRes)).

error_handling_test() ->
    Opts = setup_test_env(),
    M1 = #{},

    %% Missing key parameter
    {error, E1} = get(M1, #{}, Opts),
    ?assertEqual(400, maps:get(<<"status">>, E1)),

    %% Key not found
    {error, E2} = get(M1, #{<<"key">> => <<"nonexistent">>}, Opts),
    ?assertEqual(404, maps:get(<<"status">>, E2)).

-endif.
```
