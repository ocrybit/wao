## Testing

### HyperBEAM (Recommended)

Native Lua execution via `lua@5.3a`. No external CU needed.

```bash
HB_TIMEOUT=120 node --test --test-concurrency=1 vibe/apps/tests/hyperbeam.test.js
```

### In-Memory (ArMem)

Fast but behavior may differ from real execution.

```bash
npm test -- vibe/apps/tests/armem.test.js
```

## Lua App Pattern

```lua
-- State persists between messages
Count = Count or 0
Owner = Owner or ao.env.Process.Owner

-- Handler pattern
Handlers.add("Inc", "Inc", function(msg)
  Count = Count + 1
  msg.reply({ Data = tostring(Count) })
end)

Handlers.add("Get", "Get", function(msg)
  msg.reply({ Data = tostring(Count) })
end)
```

# Manual Development Guide

For developers who prefer coding without AI assistance.

## Environment Setup

```bash
# Clone and setup
git clone https://github.com/ocrybit/wao.git
cd wao
npm install

# Install Erlang (for HyperBEAM)
asdf plugin add erlang
asdf install erlang 27.3.4.6
asdf global erlang 27.3.4.6

# Install rebar3
asdf plugin add rebar
asdf install rebar 3.26.0
asdf global rebar 3.26.0

# Configure HyperBEAM
cat > .env.hyperbeam << 'EOF'
ARWEAVE_GATEWAY=https://arweave-proxy.ocrybit.workers.dev
HB_REBAR3=false
EOF
```

## Development Workflow

### Lua App Development

```
┌─────────────────────────────────────────────────────────────────┐
│                    Lua App Development Flow                      │
├─────────────────────────────────────────────────────────────────┤
│   1. WRITE          2. TEST IN-MEMORY       3. TEST HYPERBEAM   │
│   ─────────         ─────────────────       ─────────────────   │
│   app.lua      ───► ArMem (fast)       ───► HyperBEAM (prod)    │
│                                                                  │
│   • Handlers        • npm test              • Native Lua exec   │
│   • State           • ~14 seconds           • ~25 seconds       │
│   • Logic           • Full WASM             • luerl runtime     │
└─────────────────────────────────────────────────────────────────┘
```

**Step 1: Write Lua Code**
```lua
-- vibe/apps/myapp.lua
Count = Count or 0

Handlers.add("Inc", "Inc", function(msg)
  Count = Count + 1
  msg.reply({ Data = tostring(Count) })
end)

Handlers.add("Get", "Get", function(msg)
  msg.reply({ Data = tostring(Count) })
end)
```

**Step 2: Write Tests**
```javascript
// vibe/apps/tests/myapp.test.js
import { describe, it } from 'node:test'
import assert from 'assert'
import { readFileSync } from 'fs'
import { ArMem, connect, acc, scheduler } from '../../../src/test.js'

const luaCode = readFileSync('vibe/apps/myapp.lua', 'utf-8')

describe('MyApp', () => {
  it('should increment', async () => {
    const mem = new ArMem()
    const { spawn, message, dryrun } = connect(mem)

    const pid = await spawn({
      signer: acc[0].signer,
      scheduler,
      module: mem.modules.aos2_0_6,
    })

    await message({
      process: pid,
      signer: acc[0].signer,
      tags: [{ name: 'Action', value: 'Eval' }],
      data: luaCode,
    })

    await message({
      process: pid,
      signer: acc[0].signer,
      tags: [{ name: 'Action', value: 'Inc' }],
    })

    const res = await dryrun({
      process: pid,
      tags: [{ name: 'Action', value: 'Get' }],
    })

    assert.equal(res.Messages[0].Data, '1')
  })
})
```

**Step 3: Run Tests**
```bash
# In-memory (fast iteration)
npm test -- vibe/apps/tests/myapp.test.js

# HyperBEAM (production validation)
. ~/.asdf/asdf.sh && HB_TIMEOUT=120 node --test --test-concurrency=1 vibe/apps/tests/hyperbeam.test.js
```

### Erlang Device Development

```
┌─────────────────────────────────────────────────────────────────┐
│                 Erlang Device Development Flow                   │
├─────────────────────────────────────────────────────────────────┤
│   1. WRITE          2. REGISTER         3. TEST & BUILD UI      │
│   ─────────         ─────────           ────────────────        │
│   dev_xxx.erl  ───► hb_opts.erl   ───►  Vitest + React          │
│                                                                  │
│   • HTTP routes     • Add to map        • HTTP client class     │
│   • persistent_term • Set ~path         • Component tests       │
│   • Business logic  • Restart HB        • Full frontend         │
└─────────────────────────────────────────────────────────────────┘
```

**Step 1: Write Erlang Device**
```erlang
%% hyperbeam/src/dev_mydevice.erl
-module(dev_mydevice).
-export([info/3, routes/0, get/3, set/3]).

info(_M1, _M2, _Opts) ->
    #{name => <<"mydevice">>, version => <<"1.0">>}.

routes() ->
    #{<<"get">> => fun get/3, <<"set">> => fun set/3}.

get(_M1, _M2, _Opts) ->
    Value = persistent_term:get({?MODULE, value}, 0),
    {ok, #{value => Value}}.

set(M1, _M2, _Opts) ->
    Value = maps:get(<<"value">>, M1, 0),
    persistent_term:put({?MODULE, value}, Value),
    {ok, #{value => Value}}.
```

**Step 2: Register Device**
```erlang
%% In hyperbeam/src/hb_opts.erl default_devices():
<<"mydevice@1.0">> => #{
    module => dev_mydevice,
    routes => dev_mydevice:routes(),
    <<"~path">> => <<"~mydevice@1.0">>
}
```

**Step 3: Create JS Client**
```javascript
// vibe/apps/mydevice-ui/src/lib/MyDeviceClient.js
export default class MyDeviceClient {
  constructor(baseUrl) {
    this.baseUrl = baseUrl
  }

  async get() {
    const res = await fetch(`${this.baseUrl}/~mydevice@1.0/get`)
    return res.json()
  }

  async set(value) {
    const res = await fetch(`${this.baseUrl}/~mydevice@1.0/set`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ value })
    })
    return res.json()
  }
}
```

**Step 4: Write Vitest Tests**
```javascript
// vibe/apps/mydevice-ui/src/test/device.test.js
import { describe, it, expect, beforeAll, afterAll } from 'vitest'
import HyperBEAM from '../../../../../src/hyperbeam.js'
import MyDeviceClient from '../lib/MyDeviceClient.js'

describe('MyDevice', () => {
  let hbeam, client

  beforeAll(async () => {
    hbeam = await new HyperBEAM({ reset: true, timeout: 120 }).ready()
    client = new MyDeviceClient(hbeam.url)
  }, 60000)

  afterAll(() => hbeam?.kill())

  it('should get and set', async () => {
    await client.set(42)
    const result = await client.get()
    expect(result.value).toBe(42)
  })
})
```

**Step 5: Run Tests**
```bash
cd vibe/apps/mydevice-ui && npm test
```
